cmake_minimum_required(VERSION 3.20)
project(CompIntel3CoreTests LANGUAGES CXX)

enable_testing()
add_executable(CompIntel3CoreTests GameTests.cpp ../src/core/Game.cpp)
target_compile_features(CompIntel3CoreTests PRIVATE cxx_std_20)
target_include_directories(CompIntel3CoreTests PRIVATE ../src/core)
if(MSVC)
    target_compile_options(CompIntel3CoreTests PRIVATE /W4 /permissive- /utf-8)
else()
    target_compile_options(CompIntel3CoreTests PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_test(NAME GameRulesAndProtocol COMMAND CompIntel3CoreTests)

if(WIN32)
    add_executable(CompIntel3GomokuExample ../examples/gomoku.cpp)
    add_executable(CompIntel3QuoridorExample ../examples/quoridor.cpp)
    target_compile_features(CompIntel3GomokuExample PRIVATE cxx_std_20)
    target_compile_features(CompIntel3QuoridorExample PRIVATE cxx_std_20)
    add_executable(CompIntel3ExampleTests ExampleTests.cpp ../src/core/Game.cpp ../src/runtime/PlayerProcess.cpp)
    target_compile_features(CompIntel3ExampleTests PRIVATE cxx_std_20)
    target_include_directories(CompIntel3ExampleTests PRIVATE ../src)
    if(MSVC)
        target_compile_options(CompIntel3ExampleTests PRIVATE /W4 /permissive- /utf-8)
        target_compile_definitions(CompIntel3ExampleTests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    endif()
    add_dependencies(CompIntel3ExampleTests CompIntel3GomokuExample CompIntel3QuoridorExample)
    add_test(NAME ExamplePlayerMatches COMMAND CompIntel3ExampleTests
        $<TARGET_FILE:CompIntel3GomokuExample> $<TARGET_FILE:CompIntel3QuoridorExample>)
    set_tests_properties(ExamplePlayerMatches PROPERTIES TIMEOUT 30)

    enable_language(RC)
    configure_file(StorageSamples.rc.in "${CMAKE_CURRENT_BINARY_DIR}/StorageSamples.rc" @ONLY)
    add_executable(CompIntel3StorageTests StorageTests.cpp ../src/ui/Storage.cpp ../src/core/Game.cpp
        "${CMAKE_CURRENT_BINARY_DIR}/StorageSamples.rc")
    target_compile_features(CompIntel3StorageTests PRIVATE cxx_std_20)
    target_include_directories(CompIntel3StorageTests PRIVATE ../src)
    target_compile_definitions(CompIntel3StorageTests PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_link_libraries(CompIntel3StorageTests PRIVATE windowsapp runtimeobject ole32 shell32)
    if(MSVC)
        target_compile_options(CompIntel3StorageTests PRIVATE /W4 /permissive- /utf-8)
    endif()
    add_test(NAME StorageRoundTripAndRecovery COMMAND CompIntel3StorageTests)
    set_tests_properties(StorageRoundTripAndRecovery PROPERTIES TIMEOUT 30)
endif()
