cmake_minimum_required(VERSION 3.24)
project(CompIntel3Core LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
enable_testing()
if(MSVC)
  add_compile_options(/W4 /utf-8 /permissive-)
  add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
add_subdirectory(tests)
if(WIN32 AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/RuntimeTests.cpp")
  add_executable(RuntimeTests tests/RuntimeTests.cpp src/runtime/PlayerProcess.cpp)
  target_include_directories(RuntimeTests PRIVATE src)
  add_test(NAME RuntimeTests COMMAND RuntimeTests)
  set_tests_properties(RuntimeTests PROPERTIES TIMEOUT 90)
endif()
